    } else if (seatGrid && seatGrid.parentElement) {
      seatGrid.parentElement.insertBefore(strip, seatGrid);
    } else if (play) {
      play.insertBefore(strip, play.firstChild);
    } else {
      document.body.appendChild(strip);
    }
    if (app && app.rail) app.rail.stripEl = strip;
    return strip;
  }

  function updateStripVisibility(app) {
    var strip = (app && app.rail && app.rail.stripEl) || document.getElementById('cq-rail-strip');
    var density = app ? densityFor(app) : 'off';
    var on = featureOn() && density !== 'off' && !isDealerShiftContext(app);
    document.body.classList.toggle('cq-rail-off', !on);
    if (!strip) return;
    strip.style.display = on ? '' : 'none';
    strip.setAttribute('aria-hidden', on ? 'false' : 'true');
  }

  function maybeWhisper(app) {
    var help = app && app.help;
    if (!help) return;
    var strip = mountStrip(app);
    var el = strip.querySelector('.cq-rail-whisper');
    if (!el) return;
    var density = densityFor(app);
    if (density === 'off') {
      el.classList.add('hidden');
      return;
    }
    if (help.railWhisper && help.railWhisper()) {
      el.textContent = 'Two other seats just took cards. Tag them.';
      el.classList.remove('hidden');
    } else if (help.level === 1) {
      el.textContent = 'Rail first — then your hand.';
      el.classList.remove('hidden');
      setTimeout(function () { el.classList.add('hidden'); }, 1600);
    } else {
      el.textContent = '';
      el.classList.add('hidden');
    }
  }

  function maybeCoachMark(app) {
    if (!app || !app.save || !app.help) return;
    ensureRailSettings(app);
    if (app.save.flags.railIntroSeen) return;
    if (app.help.level !== 0) return;
    var hp = (app.save.stats && app.save.stats.handsPlayed) || 0;
    if (hp >= 12) return;
    if (densityFor(app) === 'off') return;
    app.save.flags.railIntroSeen = true;
    if (typeof app.persist === 'function') app.persist();
    if (typeof app.toast === 'function') {
      app.toast('Those small cards still count.', 'info', 4200);
    }
  }

  function dwellMs(help) {
    if (help && typeof help.railDwellMs === 'function') return Math.max(350, help.railDwellMs());
    var lvl = help && typeof help.level === 'number' ? help.level : 4;
    return DWELL[lvl] || 350;
  }

  function shouldFlashTag(help) {
    if (!help) return false;
    if (typeof help.railShowTagLabel === 'function' && help.railShowTagLabel()) return 'label';
    if (typeof help.railShowTagPip === 'function' && help.railShowTagPip()) return 'pip';
    return false;
  }

  function shouldShowFacesAfterTuck(help) {
    if (help && typeof help.railKeepFaces === 'function') return !!help.railKeepFaces();
    return !!(help && help.level <= 1);
  }

  function cardLabel(card) {
    if (typeof formatCardLabel === 'function') return formatCardLabel(card);
    var suitMap = { S: '\u2660', H: '\u2665', D: '\u2666', C: '\u2663' };
    return String(card.rank || '') + (suitMap[card.suit] || card.suit || '');
  }
