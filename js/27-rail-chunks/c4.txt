    app.rail.dealt.push({ card: card, tag: tag, handId: handId, ms: Date.now() });
  }

  async function dealRailCardImpl(railHandId) {
    if (!this.shoe) {
      if (typeof this.ensureShoe === 'function') this.ensureShoe();
      if (!this.shoe) throw new Error('Shoe unavailable');
    }
    if (this.shoe.cardsRemaining < 1) {
      if (typeof this.ensureShoe === 'function') this.ensureShoe();
      if (this.shoe.cardsRemaining < 1) throw new Error('Shoe empty');
    }
    var card = this.shoe.deal();
    var tag = this.counter.recordCardRemovedFromShoe(card);
    this.help.recordRunningCountSnapshot(this.counter.runningCount);
    ensureAppRail(this, densityFor(this));
    if (!this.rail.hands[railHandId] && typeof Hand !== 'undefined') {
      this.rail.hands[railHandId] = new Hand();
    }
    if (this.rail.hands[railHandId] && typeof this.rail.hands[railHandId].add === 'function') {
      this.rail.hands[railHandId].add(card);
    }
    noteDealt(this, card, tag, railHandId);
    this.lastHiLoTagDealt = tag;
    try {
      if (typeof Sounds !== 'undefined' && Sounds.play) Sounds.play('card');
    } catch (e) {}
    await animateCard(card, tag, railHandId, this.help);
    return card;
  }

  function clearDealLocks(self) {
    self.dealing = false;
    self._betSubmitting = false;
    if (typeof self.unlockDealButton === 'function') self.unlockDealButton();
  }

  function refundToBet(self) {
    self.phase = 'bet';
    if (self.playerHands[0]) {
      self.bankroll += self.playerHands[0].bet;
      if (self.session) self.session.wagered = Math.max(0, self.session.wagered - self.playerHands[0].bet);
    }
    if (typeof self.render === 'function') self.render();
  }
