        await self.dealCardAnimated(self.playerHands[0].hand, 'player');
        self.hideHole = true;
        await self.dealCardAnimated(self.dealer, 'dealer hole', true);
      }

      if (timedOut) return;

      var up = self.dealer.cards[0] && self.dealer.cards[0].rank;
      if (!up) throw new Error('Dealer upcard missing');
      if (up === 'A') {
        if (typeof self.franchiseDealerEvent === 'function') self.franchiseDealerEvent('deal.upcard.ace');
        clearDealLocks(self);
        self.showInsuranceModal();
        self.render();
        return;
      }
      clearDealLocks(self);
      await self.afterInsurance();
    } catch (err) {
      if (timedOut) return;
      console.error('CountQuest: dealInitial (rail) failed —', err);
      clearDealLocks(self);

      // Soft fallback: if nothing meaningful dealt yet, try original dealInitial.
      var railDealt = (self.rail && self.rail.dealt && self.rail.dealt.length) || 0;
      var playerSize = (self.playerHands[0] && self.playerHands[0].hand && self.playerHands[0].hand.size) || 0;
      var dealerSize = (self.dealer && self.dealer.size) || 0;
      var canFallback = typeof origDeal === 'function' && railDealt === 0 && playerSize === 0 && dealerSize === 0;
      if (canFallback) {
        try {
          return await origDeal.apply(self, []);
        } catch (e2) {
          console.error('CountQuest: original dealInitial fallback failed —', e2);
          clearDealLocks(self);
        }
      }

      if (typeof self.toast === 'function') {
        self.toast('Could not complete the deal — try placing your bet again', 'error', 4000);
      }
      refundToBet(self);
    } finally {
      finished = true;
      clearTimeout(abortTimer);
      // Always unlock — even if afterInsurance path already cleared dealing.
      if (!timedOut) {
        self.dealing = false;
        self._betSubmitting = false;
        if (typeof self.unlockDealButton === 'function') self.unlockDealButton();
      }
    }
  }
