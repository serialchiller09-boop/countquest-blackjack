  function handCount(density) {
    if (density === 'busy') return 3;
    if (density === 'quiet') return 2;
    return 0;
  }

  function handIdsFor(density) {
    if (density === 'busy') return HAND_IDS_BUSY.slice();
    if (density === 'quiet') return HAND_IDS_QUIET.slice();
    return [];
  }

  function ensureAppRail(app, density) {
    if (!app.rail) {
      app.rail = { density: density || 'off', hands: {}, dealt: [], stripEl: null };
    }
    app.rail.density = density || app.rail.density || 'off';
    return app.rail;
  }

  function planRound(app) {
    var density = densityFor(app);
    var rail = ensureAppRail(app, density);
    rail.density = density;
    rail.dealt = [];
    rail.hands = {};
    var ids = handIdsFor(density);
    for (var i = 0; i < ids.length; i++) {
      rail.hands[ids[i]] = typeof Hand !== 'undefined' ? new Hand() : { cards: [], add: function (c) { this.cards.push(c); }, get size() { return this.cards.length; } };
    }
    mountStrip(app);
    updateStripVisibility(app);
    maybeWhisper(app);
    maybeCoachMark(app);
    return rail;
  }

  function resetRound(app) {
    if (!app) return;
    var rail = ensureAppRail(app, densityFor(app));
    rail.dealt = [];
    rail.hands = {};
    if (rail.stripEl) {
      var stack = rail.stripEl.querySelector('.cq-rail-stack');
      if (stack) stack.innerHTML = '';
      var fly = rail.stripEl.querySelector('.cq-rail-fly-layer');
      if (fly) fly.innerHTML = '';
      var whisper = rail.stripEl.querySelector('.cq-rail-whisper');
      if (whisper) {
        whisper.textContent = '';
        whisper.classList.add('hidden');
      }
      var countEl = rail.stripEl.querySelector('.cq-rail-seen-count');
      if (countEl) countEl.textContent = '';
    }
    updateStripVisibility(app);
  }
