        ensureRailSettings(app);
        app.save.settings.railDensity = btn.getAttribute('data-rail-density');
        if (typeof app.persist === 'function') app.persist();
        syncSettingsUI(app);
        updateStripVisibility(app);
        if (typeof app.toast === 'function') {
          app.toast('Rail cards: ' + app.save.settings.railDensity + ' (next deal)', 'info', 2200);
        }
      });
    });

  }
