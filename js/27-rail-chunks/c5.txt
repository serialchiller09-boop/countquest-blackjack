.rank;
      if (!up) throw new Error('Dealer upcard missing');
      if (up === 'A') {
        if (typeof self.franchiseDealerEvent === 'function') self.franchiseDealerEvent('deal.upcard.ace');
        self.dealing = false;
        if (typeof self.unlockDealButton === 'function') self.unlockDealButton();
        self.showInsuranceModal();
        self.render();
        return;
      }
      self.dealing = false;
      if (typeof self.unlockDealButton === 'function') self.unlockDealButton();
      await self.afterInsurance();
    } catch (err) {
      console.error('CountQuest: dealInitial (rail) failed —', err);
      if (typeof self.toast === 'function') {
        self.toast('Could not complete the deal — try placing your bet again', 'error', 4000);
      }
      self.dealing = false;
      self._betSubmitting = false;
      if (typeof self.unlockDealButton === 'function') self.unlockDealButton();
      self.phase = 'bet';
      if (self.playerHands[0]) {
        self.bankroll += self.playerHands[0].bet;
        if (self.session) self.session.wagered = Math.max(0, self.session.wagered - self.playerHands[0].bet);
      }
      self.render();
    }
  }
